// 頂点シェーダーへの移植（gl_VertexID を i として使用）
float i = float(gl_VertexID);
float m = mod(i, 16.0) * 13.0;
float k = 9.0 * cos(i*5.0) * sin(i);
float e = cos(i*3.0) * cos(i*2.0) * 9.0;
float mg = sqrt(k*k + e*e);
float d = pow(mg,3.0)/1999.0 + 1.5 - pow(sin(uT/2.0+m),3.0)/3.0;
float c = d/16.0 - uT/48.0 + m;
float p = pow(d, sin(d*d - uT + m));
vec2 pos = vec2(99.0*sin(c) + k*p + 200.0,
                 99.0*sin(c*4.0) + e*p + 200.0);
